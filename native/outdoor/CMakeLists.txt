cmake_minimum_required(VERSION 3.22)
project(CasMateOutdoor LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  add_compile_options(/utf-8 "$<$<COMPILE_LANGUAGE:CXX>:/EHsc>")
endif()
set(DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../../tmp/outdoor-native-deps")
if(NOT EXISTS "${DEPS}/srt/.casmate-sha256" OR NOT EXISTS "${DEPS}/mbedtls/.casmate-sha256")
  message(FATAL_ERROR "Run node tools/prepare-outdoor-native.mjs before configuring native builds")
endif()
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(MBEDTLS_FATAL_WARNINGS OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS}/mbedtls" mbedtls EXCLUDE_FROM_ALL)
set(ENABLE_APPS OFF CACHE BOOL "" FORCE)
set(ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(ENABLE_STATIC ON CACHE BOOL "" FORCE)
set(ENABLE_ENCRYPTION ON CACHE BOOL "" FORCE)
set(ENABLE_LOGGING OFF CACHE BOOL "" FORCE)
set(USE_ENCLIB mbedtls CACHE STRING "" FORCE)
set(SSL_INCLUDE_DIRS "${DEPS}/mbedtls/include")
set(SSL_LIBRARY_DIRS "${CMAKE_CURRENT_BINARY_DIR}/mbedtls")
set(SSL_LIBRARIES mbedcrypto)
add_subdirectory("${DEPS}/srt" srt EXCLUDE_FROM_ALL)
add_library(outdoor-core STATIC transport.cpp ts-progress.cpp)
target_include_directories(outdoor-core PUBLIC "${DEPS}/srt/srtcore" "${DEPS}/srt/common" "${CMAKE_CURRENT_BINARY_DIR}/srt" "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(outdoor-core PUBLIC srt_static)
if(WIN32)
  target_compile_definitions(outdoor-core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(outdoor-core PUBLIC ws2_32)
endif()
if(ANDROID)
  add_library(casmate-srt SHARED android-jni.cpp)
  target_link_libraries(casmate-srt PRIVATE outdoor-core log)
else()
  add_executable(casmate-outdoor-worker receiver.cpp)
  target_link_libraries(casmate-outdoor-worker PRIVATE outdoor-core)
  add_executable(outdoor-native-tests tests.cpp)
  target_link_libraries(outdoor-native-tests PRIVATE outdoor-core)
  enable_testing()
  add_test(NAME outdoor-native COMMAND outdoor-native-tests)
endif()
